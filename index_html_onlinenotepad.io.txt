    <script>

        let currentRole = "Mert";

        let davetiyeler = [

            { id: 1, adSoyad: "Ahmet Yılmaz", telefon: "05321112233", birim: "Kültür Daire", misafirSahibi: "Şebnem", koltuklar: ["A2-S4"], onayli: true },

            { id: 2, adSoyad: "Ayşe Kaya", telefon: "05059998877", birim: "Büyükşehir", misafirSahibi: "Ömür", koltuklar: ["PROT-S1"], onayli: false }

        ];



        function switchTab(tabName) {

            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

            document.querySelectorAll('.tab-btn').forEach(el => {

                el.classList.remove('bg-indigo-600', 'text-white', 'shadow');

                el.classList.add('text-gray-600');

            });



            document.getElementById(`sec-${tabName}`).classList.remove('hidden');

            const btn = document.getElementById(`tab-${tabName}`);

            btn.classList.add('bg-indigo-600', 'text-white', 'shadow');

            btn.classList.remove('text-gray-600');

        }



        function toggleUserRole() {

            currentRole = (currentRole === "Mert") ? "Şebnem" : "Mert";

            document.getElementById("currentUser").innerText = currentRole;

            renderDavetiyeListesi();

        }



        function rehberdenSec() {

            if ('contacts' in navigator && 'ContactsManager' in window) {

                navigator.contacts.select(['name', 'tel'], { multiple: false })

                .then(contacts => {

                    if (contacts.length > 0) {

                        document.getElementById('telefon').value = contacts[0].tel[0];

                        if(!document.getElementById('adSoyad').value) document.getElementById('adSoyad').value = contacts[0].name[0];

                    }

                }).catch(() => {});

            } else {

                alert("Telefondan doğrudan rehber simülasyonu: Telefon rehberi açılıyor...");

                document.getElementById('adSoyad').value = "Örnek Kişi";

                document.getElementById('telefon').value = "05320001122";

            }

        }



        function handleDavetiyeEkle(e) {

            e.preventDefault();

            const koltukSelect = document.getElementById('koltukSecim');

            const seciliKoltuklar = Array.from(koltukSelect.selectedOptions).map(opt => opt.value);



            const yeni = {

                id: Date.now(),

                adSoyad: document.getElementById('adSoyad').value,

                telefon: document.getElementById('telefon').value,

                birim: document.getElementById('birim').value,

                misafirSahibi: document.getElementById('misafirSahibi').value,

                koltuklar: seciliKoltuklar,

                onayli: currentRole === "Mert"

            };



            davetiyeler.push(yeni);

            alert(yeni.onayli ? "Davetiye eklendi ve onaylandı!" : "Davetiye eklendi. Mert onayına gönderildi.");

            document.getElementById('davetiyeForm').reset();

            renderDavetiyeListesi();

        }



        function davetiyeOnayla(id) {

            if(currentRole !== "Mert") {

                alert("Yalnızca 'Mert' kullanıcısı onaylayabilir!");

                return;

            }

            const item = davetiyeler.find(d => d.id === id);

            if(item) { item.onayli = true; renderDavetiyeListesi(); }

        }



        function sendWhatsApp(id) {

            const item = davetiyeler.find(d => d.id === id);

            if(!item) return;



            if(!item.onayli) {

                alert("Onaylanmamış davetiye WhatsApp'tan gönderilemez!");

                return;

            }



            const mesaj = `Sayın *${item.adSoyad}*, Aşık Veysel Açık Hava Tiyatrosu davetiyeniz onaylanmıştır.%0A%0A*Koltuk:* ${item.koltuklar.join(', ')}%0A*Ev Sahibi:* ${item.misafirSahibi}`;

            const tel = item.telefon.replace(/[^0-9]/g, '');

            window.open(`https://api.whatsapp.com/send?phone=90${tel}&text=${mesaj}`, '_blank');

        }



        function renderDavetiyeListesi() {

            const container = document.getElementById('davetiyeListesiMobil');

            container.innerHTML = '';



            davetiyeler.forEach(d => {

                const div = document.createElement('div');

                div.className = "p-3 border rounded-xl bg-gray-50 flex justify-between items-center text-xs";

                

                const onayKodu = d.onayli 

                    ? `<span class="text-[10px] bg-green-100 text-green-700 px-2 py-0.5 rounded-full font-bold">Onaylı</span>`

                    : `<span class="text-[10px] bg-amber-100 text-amber-700 px-2 py-0.5 rounded-full font-bold">Mert Onayı Bekliyor</span>`;



                let butonlar = '';

                if(!d.onayli && currentRole === "Mert") {

                    butonlar += `<button onclick="davetiyeOnayla(${d.id})" class="bg-green-600 text-white px-2 py-1 rounded text-[10px] font-bold mr-1">Onayla</button>`;

                }

                butonlar += `<button onclick="sendWhatsApp(${d.id})" class="bg-emerald-500 text-white px-2 py-1 rounded text-[10px] font-bold"><i class="fa-brands fa-whatsapp"></i> Gönder</button>`;



                div.innerHTML = `

                    <div class="space-y-1">

                        <div class="font-bold text-gray-900">${d.adSoyad} <span class="font-normal text-gray-500">(${d.birim})</span></div>

                        <div class="text-[10px] text-gray-600">Ev Sahibi: <strong>${d.misafirSahibi}</strong> | Koltuk: <span class="font-mono font-bold text-indigo-700">${d.koltuklar.join(', ')}</span></div>

                        <div>${onayKodu}</div>

                    </div>

                    <div class="whitespace-nowrap">${butonlar}</div>

                `;

                container.appendChild(div);

            });

        }



        function finansHesapla() {

            const tahsis = parseFloat(document.getElementById('tahsisBedeli').value) || 0;

            const ekKoltuk = parseFloat(document.getElementById('ekKoltukBedeli').value) || 0;

            const teminat = parseFloat(document.getElementById('teminatBedeli').value) || 0;

            const odenen = parseFloat(document.getElementById('odenenTutar').value) || 0;



            const toplam = tahsis + ekKoltuk;

            const kalan = toplam - odenen;



            document.getElementById('txtToplamMaliyet').innerText = toplam.toLocaleString('tr-TR') + ' ₺';

            document.getElementById('txtTeminat').innerText = teminat.toLocaleString('tr-TR') + ' ₺';

            document.getElementById('txtOdenen').innerText = odenen.toLocaleString('tr-TR') + ' ₺';

            document.getElementById('txtKalan').innerText = kalan.toLocaleString('tr-TR') + ' ₺';

        }



        function ekstraKontenjanEkle() {

            const b = prompt("Birim/Şirket İsmi:");

            const a = prompt("Koltuk Sayısı:");

            if(b && a) alert(`${b} için +${a} koltuk tanımlandı.`);

        }



        function yeniSozlesmeYukle() {

            const isim = prompt("Firma / Kişi İsmi:");

            if(isim) {

                const box = document.getElementById('sozlesmeListesi');

                box.innerHTML += `

                    <div class="p-2.5 border rounded-lg flex justify-between items-center bg-gray-50 text-xs">

                        <div>

                            <div class="font-semibold text-gray-800">${isim}</div>

                            <div class="text-[10px] text-gray-500">Yeni Ek Sözleşme</div>

                        </div>

                        <button onclick="alert('Açılıyor...')" class="text-indigo-600 font-bold text-xs bg-indigo-50 px-2 py-1 rounded">Görüntüle</button>

                    </div>

                `;

            }

        }



        window.onload = function() {

            renderDavetiyeListesi();

            finansHesapla();

        };

    </script>
